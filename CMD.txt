DO THIS IS LAPTOP TERMINAL - npm install -g pnpm

# 1. Setup Environment Configuration
cp .env.example .env

# 2. Start PostgreSQL / PostGIS Database (Make sure Docker Desktop is open)
docker compose up -d

# 3. Install Monorepo Dependencies
pnpm install

# 4. Sync Python Environment (ML Phase 1)
uv sync --directory ml/phase1/training

# 5. Start Development Server (Runs Next.js Web & NestJS API in parallel)
pnpm dev

# 6. Run Offline Demo
pnpm demo:offline

# 7. Run Phase 1 Model Inference (Oil Spill Detection)
1st Install libraries
pip install torch torchvision rasterio numpy matplotlib

# Option A: Batch script (CMD / PowerShell)
.\run_model.ps1 -image data/sample_sar_image.tif -output output_mask.tif

# Option B: Direct pnpm / python command
pnpm infer:phase1 --image <path_to_image.tif> --output output_mask.tif
# or
python ml/phase1/infer.py --image <path_to_image.tif> --model models/phase1/unet_oil_spill_v0.1.pth --output output_mask.tif

# 8. Run test of model
python -m pytest ml/phase1/tests/ -v
